### 23.1 Why separate repositories?

| Reason | Explanation |
|---|---|
| Independent release cycles | Firmware releases are rare and risky; dashboard releases are frequent and low-risk |
| Different toolchains | PlatformIO (C++), Python/pytest, Node/TypeScript, Firebase CLI |
| Access control | Firebase rules and power-control code can have stricter reviewers |
| Clear ownership | Each repo maps to one subsystem and one CODEOWNERS group |
| No duplicated code | Shared *contracts* (JSON Schema, rule IDs) are versioned in `smart-water-main/schema/`; *code* isn't shared by copy-paste |

## 24. Development Method: 17 Phases

<p align="center"><img src="diagrams/19_phase_roadmap.png" alt="Phase roadmap" width="100%"/></p>

<sub>Figure 24.1: Indicative 17-phase plan. Testing and documentation run through every phase.</sub>

### 24.1 Rules of the method

1. **One phase at a time.** A phase is complete only when its test procedure passes and its README update is merged.
2. **Every phase delivers the 14 items:** Objective · Architecture · Folder structure · Required files · Complete code · Configuration · Installation commands · Run commands · Expected output · Test procedure · Troubleshooting · Git commands · Commit message · README update.
3. **Code stays small per phase.** Each phase adds only what it needs.
4. **Tests are not optional.** A phase with no passing tests isn't done.

### 24.2 Phase overview

| Phase | Name | Primary repo | Exit criterion |
|:-:|---|---|---|
| 1 | Repository architecture | all | 8 repos scaffolded, CI green, secret scanning active |
| 2 | ESP32 sensor acquisition | esp32 | Five sensors read, `null` on disconnect, calibration interface |
| 3 | ESP32 → Pi communication | esp32, raspberry-pi | Framed, CRC-checked, ACKed frames; replay buffer |
| 4 | Pi raw-data storage | raspberry-pi | Append-only store, hash chain, triggers tested |
| 5 | Verification engine | raspberry-pi | All catalogue rules implemented + unit tested |

```cpp
enum class Param : uint8_t { PH = 0, TURBIDITY, TDS, EC, TEMPERATURE, COUNT };
```

```cpp
// sensors/AnalogProbe.cpp
#include "AnalogProbe.h"
#include <math.h>
#ifndef UNIT_TEST
#include <Arduino.h>
#endif

float AnalogProbe::median(float* v, size_t n) {
  // insertion sort: n is small (≤ 31)
  for (size_t i = 1; i < n; ++i) {
    float key = v[i]; size_t j = i;
    while (j > 0 && v[j - 1] > key) { v[j] = v[j - 1]; --j; }
    v[j] = key;
  }
  return (n % 2) ? v[n / 2] : 0.5f * (v[n / 2 - 1] + v[n / 2]);
}

#ifndef UNIT_TEST
void AnalogProbe::begin() {
  analogSetPinAttenuation(cfg_.pin, ADC_11db);
  if (cfg_.excitationPin >= 0) { pinMode(cfg_.excitationPin, OUTPUT); digitalWrite(cfg_.excitationPin, LOW); }
}

float AnalogProbe::readMillivolts() {
  if (cfg_.excitationPin >= 0) digitalWrite(cfg_.excitationPin, HIGH);
  delay(cfg_.settleMs);
  float buf[31];
  const uint8_t n = cfg_.samples > 31 ? 31 : cfg_.samples;
  for (uint8_t i = 0; i < n; ++i) { buf[i] = (float)analogReadMilliVolts(cfg_.pin); delay(5); }
  if (cfg_.excitationPin >= 0) digitalWrite(cfg_.excitationPin, LOW);
  const float mv = median(buf, n);
  if (mv < cfg_.disconnectLowMv || mv > cfg_.disconnectHighMv) return NAN;
  return mv;
}
#endif
```

```cpp
// sensors/Calibration.h
#pragma once
#include <stdint.h>
#include "Types.h"

// Linear or two-point calibration: value = slope * mv + offset
// pH typically uses 2- or 3-point buffer calibration; EC uses standard solutions.
struct CalRecord {
  float slope = 1.0f;
  float offset = 0.0f;
  uint16_t version = 0;       // incremented on every successful calibration
  uint32_t epoch = 0;         // unix time of calibration
  bool valid() const { return version > 0 && isfinite(slope) && slope != 0.0f; }
};

class Calibration {
 public:
  // Two-point solve from (mv1 -> ref1), (mv2 -> ref2). Returns false if degenerate.
  static bool solveTwoPoint(float mv1, float ref1, float mv2, float ref2, CalRecord& out);
  static float apply(const CalRecord& c, float mv) { return c.slope * mv + c.offset; }
#ifndef UNIT_TEST
  static bool load(Param p, CalRecord& out);            // NVS
  static bool store(Param p, const CalRecord& in);      // NVS (increments version)
#endif
};
```

```cpp
// sensors/Calibration.cpp
#include "Calibration.h"
#include <math.h>
#ifndef UNIT_TEST
#include <Preferences.h>
#endif

bool Calibration::solveTwoPoint(float mv1, float ref1, float mv2, float ref2, CalRecord& out) {
  if (!isfinite(mv1) || !isfinite(mv2) || fabsf(mv2 - mv1) < 1.0f) return false;  // < 1 mV apart: degenerate
  out.slope = (ref2 - ref1) / (mv2 - mv1);
  out.offset = ref1 - out.slope * mv1;
  return isfinite(out.slope) && isfinite(out.offset);
}

#ifndef UNIT_TEST
bool Calibration::load(Param p, CalRecord& out) {
  Preferences prefs;
  if (!prefs.begin("cal", true)) return false;
  const String k = String(paramKey(p));
  out.slope = prefs.getFloat((k + ".s").c_str(), 1.0f);
  out.offset = prefs.getFloat((k + ".o").c_str(), 0.0f);
  out.version = prefs.getUShort((k + ".v").c_str(), 0);
  out.epoch = prefs.getULong((k + ".t").c_str(), 0);
  prefs.end();
  return out.valid();
```

## Phase 3: ESP32 → Raspberry Pi Communication

**1. Objective.** Deliver raw frames reliably from the ESP32 to the Pi over UART with framing, CRC, ACK/NACK, retries and a replay ring buffer, without ever losing or duplicating a reading at this hop.

**2. Architecture.**


```cpp
// communication/Frame.cpp
#include "Frame.h"

namespace frame {
uint16_t crc16_ccitt(const uint8_t* d, size_t n, uint16_t crc) {
  for (size_t i = 0; i < n; ++i) {
    crc ^= (uint16_t)d[i] << 8;
    for (int b = 0; b < 8; ++b) crc = (crc & 0x8000) ? (crc << 1) ^ 0x1021 : (crc << 1);
  }
  return crc;
}

size_t encode(const uint8_t* p, size_t len, uint8_t* out, size_t cap) {
  if (len > MAX_PAYLOAD || cap < len + OVERHEAD) return 0;
  size_t i = 0;
  out[i++] = STX;
  out[i++] = VER;
  out[i++] = (uint8_t)(len >> 8);
  out[i++] = (uint8_t)(len & 0xFF);
  for (size_t k = 0; k < len; ++k) out[i++] = p[k];
  const uint16_t crc = crc16_ccitt(out + 1, 3 + len);   // over VER|LEN|payload
  out[i++] = (uint8_t)(crc >> 8);
  out[i++] = (uint8_t)(crc & 0xFF);
  out[i++] = ETX;
  return i;
}
}  // namespace frame
```

```cpp
// communication/RingBuffer.h — fixed-capacity FIFO of serialized frames (RAM or RTC/flash-backed)
#pragma once
#include <stdint.h>
#include <string.h>

template <size_t N, size_t SLOT>
class RingBuffer {
 public:
  bool push(const char* s) {                 // overwrites oldest when full (counted as dropped)
    if (count_ == N) { tail_ = (tail_ + 1) % N; --count_; ++dropped_; }
    strncpy(slots_[head_], s, SLOT - 1); slots_[head_][SLOT - 1] = '\0';
    head_ = (head_ + 1) % N; ++count_;
    return true;
  }
  const char* peek() const { return count_ ? slots_[tail_] : nullptr; }
  void pop() { if (count_) { tail_ = (tail_ + 1) % N; --count_; } }
  size_t size() const { return count_; }
  uint32_t dropped() const { return dropped_; }
 private:
  char slots_[N][SLOT] = {};
  size_t head_ = 0, tail_ = 0, count_ = 0;
  uint32_t dropped_ = 0;
};
```

```cpp
#include "PiLink.h"
#include "Frame.h"

PiLink::Result PiLink::sendOne(const char* json, const char* recordId) {
  uint8_t buf[frame::MAX_PAYLOAD + frame::OVERHEAD];
  const size_t n = frame::encode((const uint8_t*)json, strlen(json), buf, sizeof(buf));
  if (!n) return Result::TOO_LARGE;
  for (uint8_t attempt = 0; attempt <= cfg_.maxRetries; ++attempt) {
    uart_.write(buf, n);
    String line;
    if (readLine(line, cfg_.ackTimeoutMs)) {
      if (line == String("ACK ") + recordId) { stats_.acked++; return Result::OK; }
      if (line.startsWith("NACK")) stats_.nacked++;
    } else {
      stats_.timeouts++;
    }
    delay(cfg_.retryBackoffMs * (attempt + 1));
  }
  return Result::NO_ACK;
}

void PiLink::pump() {
  // Replay buffered frames oldest-first; stop at first failure to preserve order.
  while (const char* json = buffer_.peek()) {
    char rid[40];
    if (!extractRecordId(json, rid, sizeof(rid))) { buffer_.pop(); stats_.corrupt++; continue; }
    if (sendOne(json, rid) != Result::OK) break;
    buffer_.pop();
  }
}
```

```python
# swedge/receiver/framing.py
"""Stream decoder for STX|VER|LEN|payload|CRC16|ETX frames."""
from __future__ import annotations

STX, ETX, VER = 0x02, 0x03, 0x01
MAX_PAYLOAD = 1024


def crc16_ccitt(data: bytes, crc: int = 0xFFFF) -> int:
    for byte in data:
        crc ^= byte << 8
        for _ in range(8):
            crc = ((crc << 1) ^ 0x1021) if crc & 0x8000 else (crc << 1)
            crc &= 0xFFFF
    return crc


def encode(payload: bytes) -> bytes:
    if len(payload) > MAX_PAYLOAD:
        raise ValueError("payload too large")
    body = bytes([VER]) + len(payload).to_bytes(2, "big") + payload
    return bytes([STX]) + body + crc16_ccitt(body).to_bytes(2, "big") + bytes([ETX])


class FrameDecoder:
    """Feed arbitrary byte chunks; yields ('ok', payload) or ('error', reason)."""

    def __init__(self, max_payload: int = MAX_PAYLOAD):
        self.buf = bytearray()
        self.max_payload = max_payload

    def feed(self, chunk: bytes):
        self.buf.extend(chunk)
        while True:
            start = self.buf.find(bytes([STX]))
            if start < 0:
                self.buf.clear()
                return
            if start > 0:
                del self.buf[:start]
            if len(self.buf) < 4:
                return
            ver, length = self.buf[1], int.from_bytes(self.buf[2:4], "big")
            if ver != VER or length > self.max_payload:
                del self.buf[0]
                yield ("error", "bad header")
                continue
            total = 1 + 1 + 2 + length + 2 + 1
            if len(self.buf) < total:
                return
            frame = bytes(self.buf[:total])
            del self.buf[:total]
            if frame[-1] != ETX:
                yield ("error", "missing ETX")
                continue
            body, crc = frame[1:-3], int.from_bytes(frame[-3:-1], "big")
            if crc16_ccitt(body) != crc:
                yield ("error", "bad CRC")
                continue
            yield ("ok", body[3:])
```

```python
# swedge/receiver/serial_receiver.py
"""Reads frames from UART, stores raw first, verifies, ACKs. Phase 3 stores to a JSONL spool;
Phase 4 replaces the spool with the append-only SQLite store."""
from __future__ import annotations

import json
import logging
import os
from datetime import datetime, timezone

import serial  # pyserial

from .framing import FrameDecoder

log = logging.getLogger("swedge.receiver")


def utcnow_iso() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="milliseconds")


class Receiver:
    def __init__(self, port: str, baud: int, sink, timeout: float = 0.2):
        self.ser = serial.Serial(port, baud, timeout=timeout)
        self.dec = FrameDecoder()
        self.sink = sink            # callable(msg: dict, received_ts: str) -> "stored" | "duplicate"
        self.stats = {"ok": 0, "crc_err": 0, "json_err": 0, "dup": 0}

    def _reply(self, line: str):
        self.ser.write((line + "\n").encode())

    def step(self):
        chunk = self.ser.read(256)
        if not chunk:
            return
        for kind, data in self.dec.feed(chunk):
            if kind == "error":
                self.stats["crc_err"] += 1
                self._reply(f"NACK {data}")
                continue
            rx = utcnow_iso()
            try:
                msg = json.loads(data.decode("utf-8"))
                rid = msg["record_id"]
            except (ValueError, KeyError, UnicodeDecodeError) as e:
                self.stats["json_err"] += 1
                log.warning("unparseable payload: %s", e)
                self._reply("NACK json")
                continue
            outcome = self.sink(msg, rx)
            self.stats["dup" if outcome == "duplicate" else "ok"] += 1
            self._reply(f"ACK {rid}")      # duplicates are ACKed so the node stops retrying


def main():
    logging.basicConfig(level=os.environ.get("SW_LOG_LEVEL", "INFO"))
    seen: set[str] = set()
    spool = open(os.environ.get("SW_SPOOL", "spool.jsonl"), "a", buffering=1)

    def sink(msg, rx):
        if msg["record_id"] in seen:
            return "duplicate"
        seen.add(msg["record_id"])
        spool.write(json.dumps({"received_ts": rx, "msg": msg}) + "\n")
        log.info("stored %s", msg["record_id"])
        return "stored"

    r = Receiver(os.environ.get("SW_SERIAL", "/dev/serial0"), int(os.environ.get("SW_BAUD", "115200")), sink)
    while True:
        r.step()


if __name__ == "__main__":
    main()
```

## Phase 4: Raspberry Pi Raw-Data Storage

**1. Objective.** Replace the Phase 3 spool with an **append-only** SQLite store that records every raw message verbatim with SHA-256 and a per-node hash chain, and rejects UPDATE/DELETE at the database layer.

**2. Architecture.**

```text
Receiver ─► RawStore.insert(msg, received_ts)
               ├─ canonical JSON → raw_sha256
               ├─ prev_chain_hash (last row for node) → chain_hash
               ├─ INSERT raw_readings (single transaction)
               └─ returns "stored" | "duplicate"
SQLite (WAL) with triggers:  UPDATE/DELETE on raw_readings → ABORT
```

**3. Folder structure.** `src/swedge/storage/`, `database/schema.sql`, `database/migrations/`.

**4. Required files.** `storage/db.py`, `storage/raw_store.py`, `storage/chain.py` (see §13.4), `database/schema.sql` (see §10.4), `tests/test_raw_store.py`.

```python
import sqlite3
from pathlib import Path

SCHEMA = Path(__file__).resolve().parents[3] / "database" / "schema.sql"


def connect(path: str) -> sqlite3.Connection:
    con = sqlite3.connect(path, isolation_level=None, timeout=10)   # autocommit; explicit BEGIN
    con.row_factory = sqlite3.Row
    con.execute("PRAGMA journal_mode=WAL")
    con.execute("PRAGMA synchronous=FULL")        # durability over speed (power loss is expected)
    con.execute("PRAGMA foreign_keys=ON")
    con.executescript(SCHEMA.read_text())
    return con
```

```python
# swedge/storage/raw_store.py
from __future__ import annotations

import json
import sqlite3

from .chain import GENESIS, chain_hash, raw_hash


class RawStore:
    def __init__(self, con: sqlite3.Connection):
        self.con = con

    def _prev_chain(self, node_id: str) -> str:
        row = self.con.execute(
            "SELECT chain_hash FROM raw_readings WHERE node_id=? ORDER BY rowid DESC LIMIT 1",
            (node_id,),
        ).fetchone()
        return row["chain_hash"] if row else GENESIS

    def insert(self, msg: dict, received_ts: str) -> str:
        rid = msg["record_id"]
        raw = msg.get("raw", {}) or {}
        rsha = raw_hash(msg)
        try:
            self.con.execute("BEGIN IMMEDIATE")
            prev = self._prev_chain(msg["node_id"])
            self.con.execute(
                """INSERT INTO raw_readings (record_id, node_id, boot_id, seq, sensor_ids_json,
                   device_ts, received_ts, mode, ph, turbidity_ntu, tds_ppm, ec_us_cm, temperature_c,
                   raw_electrical_json, cal_version_json, power_json, comm_json, fw_version,
                   payload_json, raw_sha256, prev_chain_hash, chain_hash)
                   VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)""",
                (
                    rid, msg["node_id"], int(msg.get("boot_id", 0)), int(msg["seq"]),
                    json.dumps(msg.get("sensor_ids", {}), sort_keys=True),
                    msg["timestamp"], received_ts, msg["mode"],
                    _num(raw.get("ph")), _num(raw.get("turbidity_ntu")), _num(raw.get("tds_ppm")),
                    _num(raw.get("ec_us_cm")), _num(raw.get("temperature_c")),
                    json.dumps(msg.get("raw_electrical")), json.dumps(msg.get("cal_version")),
                    json.dumps(msg.get("power", {})), json.dumps(msg.get("comm", {})), msg.get("fw"),
                    json.dumps(msg, sort_keys=True, separators=(",", ":")),
                    rsha, prev, chain_hash(prev, rsha),
                ),
            )
            self.con.execute("COMMIT")
            return "stored"
        except sqlite3.IntegrityError:
            self.con.execute("ROLLBACK")
            return "duplicate"
        except Exception:
            self.con.execute("ROLLBACK")
            raise

    def rows_for_node(self, node_id: str):
        return self.con.execute(
            "SELECT * FROM raw_readings WHERE node_id=? ORDER BY boot_id, seq", (node_id,)
        ).fetchall()


def _num(v):
    """Store numbers as REAL; anything else (None, strings, bools) as NULL column.
    The verbatim original is always preserved in payload_json."""
    return float(v) if isinstance(v, (int, float)) and not isinstance(v, bool) else None
```

```python
# swedge/validator/config.py
"""Load and VALIDATE verification configuration. Refuses unsafe/undocumented settings."""
from __future__ import annotations

import hashlib
import json
from pathlib import Path

import yaml

PARAMS = ("ph", "turbidity_ntu", "tds_ppm", "ec_us_cm", "temperature_c")


class ConfigError(ValueError):
    pass


def load(path: str) -> tuple[dict, str]:
    cfg = yaml.safe_load(Path(path).read_text())
    for p in PARAMS:
        lo, hi = cfg["ranges"][p]
        if not (isinstance(lo, (int, float)) and isinstance(hi, (int, float))):
            raise ConfigError(f"ranges.{p} still TBD — set from the installed sensor's datasheet")
        if lo >= hi:
            raise ConfigError(f"ranges.{p}: low must be < high")
        if cfg["max_rate_per_min"][p] <= 0:
            raise ConfigError(f"max_rate_per_min.{p} must be > 0")
    bands = {}
    for p, b in (cfg.get("attention_bands") or {}).items():
        if b and b.get("enabled"):
            if not b.get("source"):
                raise ConfigError(f"attention_bands.{p} enabled without a documented source")
            bands[p] = [b.get("low"), b.get("high")]
        else:
            bands[p] = None
    cfg["attention_bands"] = bands
    if cfg.get("tds_ec_factor") in (None, "TBD"):
        raise ConfigError("tds_ec_factor is TBD — determine for your site")
    digest = hashlib.sha256(json.dumps(cfg, sort_keys=True).encode()).hexdigest()[:16]
    return cfg, digest
```

```python
from __future__ import annotations

import json
import sqlite3


class ResultStore:
    def __init__(self, con: sqlite3.Connection, config_hash: str):
        self.con, self.config_hash = con, config_hash

    def insert(self, result) -> str:
        d = result.to_dict()
        rid = f"{d['record_id']}#v{d['engine_version']}"
        try:
            self.con.execute(
                """INSERT INTO verification_results (result_id, record_id, node_id, status, verification,
                   rules_json, raw_snapshot_json, raw_sha256, engine_version, config_hash, verified_ts)
                   VALUES (?,?,?,?,?,?,?,?,?,?,?)""",
                (rid, d["record_id"], d["node_id"], d["status"], d["verification"],
                 json.dumps(d["rules_triggered"]), json.dumps(d["raw_snapshot"]), d["raw_sha256"],
                 d["engine_version"], self.config_hash, d["verified_at"]),
            )
            return rid
        except sqlite3.IntegrityError:
            return rid        # already verified by this engine version (idempotent)
```

```python
# swedge/pipeline.py — glue used by the receiver sink
from __future__ import annotations

from datetime import datetime

from swedge.validator.engine import VerificationEngine, Status


class Pipeline:
    def __init__(self, raw_store, result_store, events, outbox, engine: VerificationEngine):
        self.raw, self.results, self.events, self.outbox, self.engine = raw_store, result_store, events, outbox, engine

    def handle(self, msg: dict, received_ts: str) -> str:
        outcome = self.raw.insert(msg, received_ts)           # 1) RAW FIRST
        if outcome == "duplicate":
            return outcome
        result = self.engine.verify(msg, datetime.fromisoformat(received_ts))   # 2) VERIFY
        result_id = self.results.insert(result)               # 3) STORE RESULT
        for hit in result.rules_triggered:                    # 4) EVENTS
            self.events.from_rule_hit(msg["node_id"], msg["record_id"], hit)
        self.outbox.enqueue("readings", msg["record_id"], {**msg, "received_ts": received_ts,
                            "verification": result.to_dict()})          # 5) QUEUE (Phase 6)
        return outcome
```
